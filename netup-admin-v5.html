function doPost(e) {
  var ss = SpreadsheetApp.openById('SHEETS_ID');
  var sheet = ss.getSheetByName('Questions') || ss.insertSheet('Questions');
  var data = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID','Soru','A','B','C','D','E','Doğru','Ders','Zorluk','Durum','Açıklama','Tarih']);
  }
  data.forEach(function(q) {
    sheet.appendRow([q.id,q.question_text,q.option_a,q.option_b,q.option_c,q.option_d,q.option_e||'',q.correct_option,q.subject,q.difficulty,q.status,q.canonical_explanation||'',q.created_at]);
  });
  return ContentService.createTextOutput(JSON.stringify({ok:true,count:data.length})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e){return doPost(e);}
